// CalcInterface.java

import java.rmi.*;

public interface CalcInterface extends Remote {
    double calculate(double n1, double n2, String operator) throws RemoteException;
}


// CalcImpl.java

import java.rmi.*;
import java.rmi.server.*;

public class CalcImpl extends UnicastRemoteObject implements CalcInterface {
    protected CalcImpl() throws RemoteException {
        super();
    }

    public double calculate(double n1, double n2, String operator) throws RemoteException {
        double result = 0;
        switch (operator) {
            case "+": result = n1 + n2; break;
            case "-": result = n1 - n2; break;
            case "*": result = n1 * n2; break;
            case "/": result = n1 / n2; break;
        }
        return result;
    }
}


// CalcServer.java

import java.rmi.*;
import java.rmi.registry.*;

public class CalcServer {
    public static void main(String[] args) {
        try {
            CalcImpl obj = new CalcImpl();
            Registry registry = LocateRegistry.createRegistry(1099);
            registry.rebind("CalcService", obj);
            System.out.println("Calculator Server is ready");
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}


// CalcClient.java

import java.rmi.*;
import java.util.Scanner;

public class CalcClient {
    public static void main(String[] args) {
        try {
            CalcInterface stub = (CalcInterface) Naming.lookup("rmi://localhost/CalcService");
            Scanner sc = new Scanner(System.in);

            System.out.print("Enter first number: ");
            double n1 = sc.nextDouble();
            System.out.print("Enter operator (+,-,*,/): ");
            String op = sc.next();
            System.out.print("Enter second number: ");
            double n2 = sc.nextDouble();

            double result = stub.calculate(n1, n2, op);
            System.out.println("Result: " + result);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
