EXPERIMENT NO. 1

Write an AWT program to create check boxes for different courses of a
university such that the courses selected will be displayed.

PROGRAM:

import java.awt.*;
import java.awt.event.*;

public class CourseCheckbox extends Frame implements ItemListener {
    Checkbox c1, c2, c3, c4;
    Label l1;

    CourseCheckbox() {
        setTitle("University Courses");
        setLayout(new FlowLayout());

        // Create Checkboxes
        c1 = new Checkbox("B.Tech");
        c2 = new Checkbox("BCA");
        c3 = new Checkbox("MCA");
        c4 = new Checkbox("MBA");

        // Create Label
        l1 = new Label("Selected Courses: None");

        // Add Item Listeners
        c1.addItemListener(this);
        c2.addItemListener(this);
        c3.addItemListener(this);
        c4.addItemListener(this);

        // Add Components
        add(c1);
        add(c2);
        add(c3);
        add(c4);
        add(l1);

        // Frame Properties
        setSize(350, 200);
        setVisible(true);

        // Close Window
        addWindowListener(new WindowAdapter() {
            public void windowClosing(WindowEvent e) {
                dispose();
            }
        });
    }

    public void itemStateChanged(ItemEvent e) {
        String courses = "";
        if (c1.getState())
            courses += "B.Tech ";
        if (c2.getState())
            courses += "BCA ";
        if (c3.getState())
            courses += "MCA ";
        if (c4.getState())
            courses += "MBA ";
        if (courses.equals(""))
            courses = "None";
        l1.setText("Selected Courses: " + courses);
    }

    public static void main(String[] args) {
        new CourseCheckbox();
    }
}
