EXPERIMENT NO. 2

Write an applet to create a list of departments of a college such that
if one of the departments is clicked, then it will be displayed on the
text field.

PROGRAM:

import java.applet.*;
import java.awt.*;
import java.awt.event.*;

/*
<applet code="DeptList.class" width="350" height="200">
</applet>
*/

public class DeptList extends Applet implements ItemListener {
    List list;
    TextField tf;
    Label l1;

    public void init() {
        setLayout(new FlowLayout());

        // Create Label
        l1 = new Label("Select a Department:");

        // Create List of Departments
        list = new List(5, false);
        list.add("Computer Science");
        list.add("Information Technology");
        list.add("Electronics & Communication");
        list.add("Mechanical Engineering");
        list.add("Civil Engineering");

        // Create TextField
        tf = new TextField(25);

        // Add Item Listener
        list.addItemListener(this);

        // Add Components
        add(l1);
        add(list);
        add(tf);
    }

    public void itemStateChanged(ItemEvent e) {
        tf.setText(list.getSelectedItem());
    }
}

HTML FILE (DeptList.html) TO RUN THE APPLET:

<html>
<body>
<applet code="DeptList.class" width="350" height="200">
</applet>
</body>
</html>

COMMANDS TO COMPILE AND RUN:

javac DeptList.java
appletviewer DeptList.html
